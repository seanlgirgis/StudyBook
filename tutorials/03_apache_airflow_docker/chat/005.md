# ============================================================
# Topic   : Apache Airflow (Docker) for Data Engineers
# File    : studybook_04_taskflow_dynamic_mapping.py
# Covers  : TaskFlow API, XComArg, dynamic task mapping
# Prereqs : Airflow Docker stack running at http://localhost:8088
# Deploy  : Place in docker/dags/ so it mounts to /opt/airflow/dags/
# Trigger : Use Airflow UI or trigger manually
# ============================================================

"""
Modern Airflow patterns in Docker.

DAGs:
  1. studybook_04_taskflow
  2. studybook_04_dynamic_mapping

Core ideas:
- @dag and @task reduce boilerplate.
- Task return values become XCom automatically.
- Dynamic mapping creates one task instance per input item.
- Keep XCom payloads small: pass file paths and metadata, not full datasets.
"""

from airflow.decorators import dag, task
from datetime import datetime
from pathlib import Path
import json
import logging
import time

OUTPUT_DIR = Path("/opt/airflow/dags/.studybook_runtime/04_taskflow_dynamic_mapping")

log = logging.getLogger("airflow.task")


@dag(
    dag_id="studybook_04_taskflow",
    description="Modern TaskFlow API pipeline",
    start_date=datetime(2024, 1, 1),
    schedule=None,
    catchup=False,
    tags=["studybook", "docker", "taskflow"],
)
def taskflow_pipeline():
    """
    TaskFlow version of extract → validate → transform → load.

    Instead of PythonOperator wiring, each Python function becomes a task.
    Airflow automatically passes return values through XCom.
    """

    @task
    def extract() -> dict:
        OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

        records = [
            {"id": i, "sensor": f"s{i % 5}", "value": i * 1.5}
            for i in range(200)
        ]

        path = OUTPUT_DIR / "taskflow_extract.json"
        path.write_text(json.dumps(records, indent=2))

        return {
            "record_count": len(records),
            "file_path": str(path),
            "source": "taskflow_simulated_api",
        }

    @task
    def validate(extract_result: dict) -> dict:
        count = extract_result["record_count"]

        if count < 100:
            raise ValueError(f"Too few records: {count}")

        logging.getLogger("airflow.task").info(
            f"Validation passed: {count} records"
        )

        return extract_result

    @task
    def transform(validated: dict) -> dict:
        input_path = Path(validated["file_path"])
        records = json.loads(input_path.read_text())

        transformed = []
        for row in records:
            value = row["value"]
            transformed.append(
                {
                    **row,
                    "value_doubled": value * 2,
                    "status": "ALERT" if value > 250 else "NORMAL",
                }
            )

        output_path = OUTPUT_DIR / "taskflow_transform.json"
        output_path.write_text(json.dumps(transformed, indent=2))

        return {
            "record_count": len(transformed),
            "file_path": str(output_path),
        }

    @task
    def load(transform_result: dict) -> None:
        logging.getLogger("airflow.task").info(
            f"Loaded {transform_result['record_count']} records "
            f"from {transform_result['file_path']}"
        )

    raw = extract()
    valid = validate(raw)
    transformed = transform(valid)
    load(transformed)


taskflow_dag = taskflow_pipeline()


@dag(
    dag_id="studybook_04_dynamic_mapping",
    description="Dynamic task mapping: one task instance per file",
    start_date=datetime(2024, 1, 1),
    schedule=None,
    catchup=False,
    tags=["studybook", "docker", "dynamic-mapping"],
)
def dynamic_mapping_pipeline():
    """
    Dynamic mapping lets Airflow create task instances at runtime.

    Instead of manually writing:
      process_file_1
      process_file_2
      process_file_3

    Airflow does:
      process_file.expand(file_path=file_list)

    This is common in real data engineering:
    - one task per landed file
    - one task per table
    - one task per partition
    - one task per customer/batch
    """

    @task
    def create_input_files() -> list[str]:
        input_dir = OUTPUT_DIR / "dynamic_inputs"
        input_dir.mkdir(parents=True, exist_ok=True)

        file_paths = []

        for file_num in range(1, 6):
            rows = [
                {
                    "file_num": file_num,
                    "row_num": row_num,
                    "value": file_num * row_num,
                }
                for row_num in range(1, 101)
            ]

            path = input_dir / f"input_file_{file_num}.json"
            path.write_text(json.dumps(rows, indent=2))
            file_paths.append(str(path))

        logging.getLogger("airflow.task").info(
            f"Created {len(file_paths)} input files"
        )

        return file_paths

    @task
    def process_file(file_path: str) -> dict:
        path = Path(file_path)
        rows = json.loads(path.read_text())

        time.sleep(1)

        total_value = sum(row["value"] for row in rows)

        output_dir = OUTPUT_DIR / "dynamic_outputs"
        output_dir.mkdir(parents=True, exist_ok=True)

        output_path = output_dir / f"processed_{path.name}"
        output_path.write_text(
            json.dumps(
                {
                    "source_file": file_path,
                    "row_count": len(rows),
                    "total_value": total_value,
                    "status": "ok",
                },
                indent=2,
            )
        )

        return {
            "source_file": file_path,
            "output_file": str(output_path),
            "row_count": len(rows),
            "total_value": total_value,
        }

    @task
    def summarize(results: list[dict]) -> None:
        total_files = len(results)
        total_rows = sum(item["row_count"] for item in results)
        total_value = sum(item["total_value"] for item in results)

        logging.getLogger("airflow.task").info(
            f"Dynamic mapping summary: "
            f"files={total_files}, rows={total_rows}, total_value={total_value}"
        )

        for item in results:
            logging.getLogger("airflow.task").info(
                f"Processed {item['source_file']} → {item['output_file']}"
            )

    files = create_input_files()
    results = process_file.expand(file_path=files)
    summarize(results)


dynamic_mapping_dag = dynamic_mapping_pipeline()
