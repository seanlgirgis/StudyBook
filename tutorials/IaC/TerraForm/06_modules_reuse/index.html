Terraform Modules and Reuse Tutorial

This page explains reusable Terraform module patterns with a local-safe example only.

Visual Learning Path

Copyable Commands

terraform version
terraform init
terraform fmt
terraform validate
terraform plan

Root main.tf

terraform {
  required_version = ">= 1.4.0"
}

module "standard_name" {
  source = "./modules/naming_standard"
  environment      = var.environment
  application_name = var.application_name
  owner            = var.owner
}

resource "terraform_data" "module_demo" {
  input = {
    standard_name = module.standard_name.name_prefix
    tags          = module.standard_name.common_tags
    lesson        = "Modules package reusable Terraform patterns."
  }
}

Root variables.tf

variable "environment" { type = string default = "dev" }
variable "application_name" { type = string default = "learnterraform" }
variable "owner" { type = string default = "Sean" }

Root outputs.tf

output "standard_name_from_module" { value = module.standard_name.name_prefix }
output "standard_tags_from_module" { value = module.standard_name.common_tags }
output "module_demo_output" { value = terraform_data.module_demo.output }

Child module files

# modules/naming_standard/main.tf
locals {
  name_prefix = "${var.environment}-${var.application_name}"
  common_tags = {
    environment = var.environment
    application = var.application_name
    owner       = var.owner
    managed_by  = "terraform"
    pattern     = "module-demo"
  }
}
# modules/naming_standard/variables.tf
variable "environment" { type = string }
variable "application_name" { type = string }
variable "owner" { type = string }
# modules/naming_standard/outputs.tf
output "name_prefix" { value = local.name_prefix }
output "common_tags" { value = local.common_tags }

Common Mistakes

  • wrong module source path
  • missing child module variables
  • referencing wrong output path
  • confusing root module and child module responsibilities
