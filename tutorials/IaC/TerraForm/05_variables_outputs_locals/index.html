Terraform Variables, Outputs, and Locals Tutorial

This page teaches reusable Terraform input/internal/output patterns using local-safe examples only.

Visual Learning Path

Copyable Commands

terraform version
terraform init
terraform fmt
terraform validate
terraform plan
terraform plan -var-file="dev.tfvars.example"
terraform plan -var-file="prod.tfvars.example"

main.tf

terraform {
  required_version = ">= 1.4.0"
}

locals {
  name_prefix = "-"
  common_tags = {
    environment = var.environment
    application = var.application_name
    owner       = var.owner
    managed_by  = "terraform"
  }
  workflow_message = "Planning  for "
}

resource "terraform_data" "variables_demo" {
  input = {
    name_prefix = local.name_prefix
    tags        = local.common_tags
    message     = local.workflow_message
  }
}

variables.tf

variable "environment" { type = string default = "dev" }
variable "application_name" { type = string default = "learnterraform" }
variable "owner" { type = string default = "Sean" }
variable "enable_extra_message" { type = bool default = true }
variable "allowed_regions" { type = list(string) default = ["us-east-1","us-east-2"] }
variable "demo_settings" {
  type = object({ size = string, retention = number })
  default = { size = "small", retention = 7 }
}
variable "example_sensitive_token" { type = string default = "do-not-use-real-secrets" sensitive = true }

tfvars Examples

terraform.tfvars.example
environment      = "dev"
application_name = "learnterraform"
owner            = "Sean"
dev.tfvars.example
environment = "dev"
demo_settings = { size = "small", retention = 7 }
prod.tfvars.example
environment = "prod"
demo_settings = { size = "medium", retention = 30 }

outputs.tf

output "name_prefix" { value = local.name_prefix }
output "common_tags" { value = local.common_tags }
output "workflow_message" { value = local.workflow_message }
output "demo_resource_output" { value = terraform_data.variables_demo.output }
output "sensitive_token_example" { value = var.example_sensitive_token sensitive = true }

Common Mistakes

  • Hardcoding environment values everywhere
  • Confusing variables (input) and locals (internal computed values)
  • Exposing sensitive outputs casually
  • Treating tfvars as secret storage without controls
