LC 416 — Partition Equal Subset Sum

0/1 Subset Sum DP  ·  canReach[i][s] = can first i elements reach sum s?

canReach[0][0] = True              // base: no elements → only sum 0 reachable
canReach[0][s>0] = False
canReach[i][s] = skip or take
  where  skip = canReach[i-1][s]                         // don't use nums[i-1]
        take = canReach[i-1][s - nums[i-1]] if s ≥ nums[i-1]  // use it (row i-1 enforces 0/1)
