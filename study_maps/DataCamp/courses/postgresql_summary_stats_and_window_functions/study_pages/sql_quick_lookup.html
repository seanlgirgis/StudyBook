Ranking

ROW_NUMBER() OVER (ORDER BY value DESC)
RANK()       OVER (ORDER BY value DESC)
DENSE_RANK() OVER (ORDER BY value DESC)

Previous and next row

LAG(value)  OVER (PARTITION BY group_col ORDER BY time_col)
LEAD(value) OVER (PARTITION BY group_col ORDER BY time_col)

Running total

SUM(value) OVER (
  PARTITION BY group_col
  ORDER BY time_col
  ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
)

Moving average

AVG(value) OVER (
  ORDER BY time_col
  ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
)
